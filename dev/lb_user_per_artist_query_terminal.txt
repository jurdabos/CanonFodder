import pandas as pd
from pathlib import Path
PQ_DIR = Path('C:/acidvuca/canonfodder/PQ')
scrobbles = pd.read_parquet(PQ_DIR / 'forevian_scrobble.parquet')
artist_info = pd.read_parquet(PQ_DIR / 'artist_info.parquet')
countries = pd.read_parquet(PQ_DIR / 'c.parquet')
df = scrobbles.merge(artist_info[['artist_name', 'country']], on='artist_name', how='left')
df = df.merge(countries[['ISO-2', 'en_name']], left_on='country', right_on='ISO-2', how='left')
artist_counts = df.groupby(['en_name', 'artist_name']).size().reset_index(name='play_count').dropna(subset=['en_name'])
top_10 = artist_counts.sort_values(['en_name', 'play_count'], ascending=[True, False]).groupby('en_name').head(10)
for country, group in top_10.groupby('en_name'):
    print(f'\n=== {country} ===')
    for _, row in group.iterrows():
        print(f"  {row['artist_name']}: {row['play_count']}")
"@
